5.5 HCL: variables, outputs, resources, data

This course uses the Docker provider so everything runs locally with no cloud bill — the concepts are identical to AWS or Azure providers, only the resource names change.

terraform/main.tf:

terraform {
  required_version = ">= 1.6"
  required_providers {
    docker = {
      source  = "kreuzwerker/docker"
      version = "~> 3.0"
    }
  }
}

provider "docker" {}

# a network, so containers can find each other by name
resource "docker_network" "app" {
  name = "${var.project}-net"
}

# a named volume for redis persistence
resource "docker_volume" "redis_data" {
  name = "${var.project}-redis-data"
}

# pull the image (data-like resource: describes something external)
resource "docker_image" "redis" {
  name         = "redis:7-alpine"
  keep_locally = true
}

resource "docker_container" "redis" {
  name  = "${var.project}-redis"
  image = docker_image.redis.image_id

  command = ["redis-server", "--appendonly", "yes"]

  networks_advanced {
    name    = docker_network.app.name
    aliases = ["redis"]
  }

  volumes {
    volume_name    = docker_volume.redis_data.name
    container_path = "/data"
  }

  restart = "unless-stopped"
}

terraform/variables.tf:

variable "project" {
  description = "Name prefix for all resources"
  type        = string
  default     = "urlshort"
}

variable "api_port" {
  description = "Host port to expose the API on"
  type        = number
  default     = 8000

  validation {
    condition     = var.api_port > 1024 && var.api_port < 65536
    error_message = "api_port must be an unprivileged port."
  }
}

variable "api_image" {
  description = "Fully qualified image, ideally tagged with a commit SHA"
  type        = string
  default     = "urlshort:0.1.0"
}

terraform/outputs.tf — outputs are how Terraform hands facts to the next tool:

output "api_url" {
  description = "Where the API is reachable"
  value       = "http://localhost:${var.api_port}"
}

output "network_name" {
  value = docker_network.app.name
}

Dependencies are inferred, not declared

Because docker_container.redis references docker_network.app.name, Terraform knows the network must exist first. You almost never write explicit ordering — it builds a dependency graph from the references and parallelises everything unrelated. Use depends_on only for dependencies that exist in reality but not in the code.
